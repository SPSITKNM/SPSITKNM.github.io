PRO :: Big O Notation · C#
// O(1) — konštantná zložitosť
int first = arr[0]; // vždy 1 operácia

// O(n) — lineárna zložitosť
void Print(int n) {
  for (int i=0; i<n; i++)
    Console.WriteLine(i);
}

// O(n²) — kvadratická zložitosť
void Pairs(int n) {
  for (int i=0; i<n; i++)
    for (int j=0; j<n; j++)
      Console.WriteLine($"({i},{j})");
}
